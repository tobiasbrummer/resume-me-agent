% Layout-Huelle einer Bewerbung.
%
% Wird vom application-lokalen bewerbung.tex via \input geladen.
% Erwartet im aktuellen Verzeichnis:
%   meta.tex         -- \def\meta...-Variablen (Empfaenger, Datum, Betreff, ...)
%   anschreiben.md   -- Anschreiben-Body
%   profil.md        -- Profil-Absatz (genau einer; wird neben dem Foto gesetzt)
%   experience.md    -- Berufserfahrung + Teaching + Volunteering + Education
%   projects.md      -- Projekte/Research (DL mit GitHub-Links)
%   languages.md     -- Sprachen
%   skills.tex       -- Skills-Block im Flex-Layout (via "## Skills" Hook)
%   sections.tex     -- optional: Section-Reihenfolge inkl. \clearpage
% tokens.tex (Design-Tokens) wird von preamble.tex geladen, nicht hier.
% Asset-Pfade (signatur, portrait) werden via TEXINPUTS aus assets/ gefunden.

\begin{document}

% Schatten-Offsets (tikz frisst \dimexpr nicht direkt in Koordinaten, darum
% als feste Laengen aus der Space-Skala). Betreff/CV-Titel: \spaceHair (0.40mm),
% Foto: \spaceXs (0.71mm).
\newlength{\schattenklein}\setlength{\schattenklein}{\spaceHair}
\newlength{\schattengross}\setlength{\schattengross}{\spaceXs}

% --- Sprachwahl (de oder en) ---
\ifdefined\metaspracheen
  \selectlanguage{english}%
\else
  \selectlanguage{ngerman}%
\fi

% Flattersatz ab hier global -- auch fuer die Absenderzeile. Sonst steht sie im
% Anschreiben im Blocksatz, im Lebenslauf (erbt \RaggedRight) im Flattersatz
% -> unterschiedliche Ausrichtung der \\-umbrochenen Zeilen.
\RaggedRight

% =============================================================
% SEITE 1 -- ANSCHREIBEN (keine Seitenzahl, zaehlt nicht mit)
% =============================================================

\thispagestyle{anschreiben}% Anschreiben: keine Seitenzahl, aber Anlagen im Footer (gleiche Baseline wie CV-Seitenzahl)

% Absenderzeile oben
{\absenderzeile}\\[\spaceFourxl]

% Empfaengerblock (linksbuendig)
\fontsizebody \metaempfaengerfirma\\
\ifx\metaempfaengername\empty\else\metaempfaengername\\\fi
\metaempfaengerstrasse\\
\metaempfaengerplzort

\vspace{\spaceXl}

% Betreff (fett, links, mit hartem purple Schatten analog Foto) und Datum (rechts).
% Text wird zweimal gesetzt: Schatten in purple bei \schattenklein, Original bei (0,0).
\noindent\hspace{-\spaceBase}%
\begin{tikzpicture}[baseline=(t.base), inner sep=0pt, outer sep=0pt]
  \node[anchor=base west, text=purple] at (\schattenklein,-\schattenklein)
        {\fontsemibold\fontsizebetreff\metabetreff};
  \node[anchor=base west]              (t) at (0,0)
        {\fontsemibold\fontsizebetreff\metabetreff};
\end{tikzpicture}%
\ifx\metabetreffzusatz\empty\else\metabetreffzusatz\fi
\hfill\fontsizeabsender\textcolor{muted}\metadatum

\vspace{\glueexpr \spaceMd plus \spaceXs minus \spaceXs\relax}

% Anrede
{\fontsizebody \metaanrede}

% Anschreiben-Body (direkt aus Markdown). Das \par MUSS innerhalb der
% \fontsizebody-Gruppe stehen, sonst nutzt der Linebreak des letzten Absatzes
% den default \baselineskip (kleiner) statt 1.2\rem -> letzte Zeile gestaucht.
% (\RaggedRight steht jetzt global oben nach der Sprachwahl.)
{\fontsizebody \markdownInput{anschreiben.md}\par}

% Grussformel + Signatur
{\fontsizebody \metagrussformel}

\vspace{-\spaceSm}
\hspace{-\spaceTwoxl}\includegraphics[height=\glueexpr \spaceFivexl plus \spaceBase minus \spaceMd\relax]{signatur}\\[\dimexpr -\spaceXs\relax]
\metaname

% Anlagen-Hinweis steht jetzt im Footer (Pagestyle "anschreiben", definiert in
% preamble.tex) -- auf gleicher Baseline wie die CV-Seitenzahl, kein \vfill noetig.

% =============================================================
% SEITE 2+ -- LEBENSLAUF (Seitenzahlen ab 1, linksbuendig im Footer)
% =============================================================

\clearpage
\setcounter{page}{1}% CV-Seitenzaehler bei 1 starten
\pagestyle{scrheadings}% ab hier Footer mit Seitenzahl (linksbuendig via preamble)
\fontsizebody% CV-default font + baselineskip (sonst rendern \markdownInput-Bloecke
            % mit 10.5pt mainfont und kleinerem baselineskip -> mehrzeilige Bullet-Items
            % wirken gestaucht). Section-Headings haben eigene \addtokomafont-Settings.

% Absenderzeile oben
{\absenderzeile}\\[\spaceFourxl]

% Platzhalter in Hoehe des Empfaengerblocks (unsichtbar via \phantom):
% haelt den CV-Titel auf gleicher Hoehe wie den Betreff im Anschreiben,
% ohne die Empfaenger-Adresse im Lebenslauf sichtbar zu setzen. Gleiche
% \\-Struktur wie der echte Block -> identische vertikale Hoehe.
\fontsizebody \phantom{\metaempfaengerfirma}\\
\ifx\metaempfaengername\empty\else\phantom{\metaempfaengername}\\\fi
\phantom{\metaempfaengerstrasse}\\
\phantom{\metaempfaengerplzort}

\vspace{\spaceXl}

% CV-Titel (sprachabhaengig via \cvtitle aus preamble.tex)
\noindent\hspace{-\spaceBase}%
\begin{tikzpicture}[baseline=(t.base), inner sep=0pt, outer sep=0pt]
  \node[anchor=base west, text=purple] at (\schattenklein,-\schattenklein)
        {\fontsemibold\fontsizebetreff\cvtitle};
  \node[anchor=base west]              (t) at (0,0)
        {\fontsemibold\fontsizebetreff\cvtitle};
\end{tikzpicture}%
\vspace{-\spaceXl}

% Profil + Portrait: Foto RECHTS, Text links. Bildhoehe = Profil-Boxhoehe.
\IfFileExists{profil.md}{%
  \begin{lrbox}{\profilbox}%
    \begin{minipage}[t]{\dimexpr\linewidth-\bildbreite\relax}\vspace{0pt}%
      \fontsizebody\markdownInput{profil.md}%
    \end{minipage}%
  \end{lrbox}%
  % \hspace nach \hfill schiebt das Bild in den rechten Seitenrand. Erzeugt
  % eine harmlose "Overfull \hbox"-Warnung im Log.
  \noindent\usebox{\profilbox}\hfill\hspace*{\spaceThreexl}%
  \begin{minipage}[t]{\bildbreite}\vspace{0pt}%
    % Foto mit hartem purple Schatten via TikZ (\schattengross rechts+unten).
    % Schatten zuerst (anchor=north west), Foto drueber bei (0,0).
    \sbox\imgbox{\includegraphics[height=\dimexpr\ht\profilbox+\dp\profilbox\relax,
                                  width=\bildbreite, keepaspectratio]{portrait}}%
    \begin{tikzpicture}[inner sep=0pt, outer sep=0pt]
      \node[anchor=north west, fill=purple,
            minimum width=\wd\imgbox,
            minimum height=\dimexpr\ht\imgbox+\dp\imgbox\relax] at (\schattengross,-\schattengross) {};
      \node[anchor=north west] at (0,0) {\usebox\imgbox};
    \end{tikzpicture}%
  \end{minipage}\par
  \vspace{\spaceMd}%
}{}%

% Section-Reihenfolge: per Application override via sections.tex (optional).
% Wenn applications/<job>/sections.tex existiert, wird das geladen statt der
% default-order unten. So kann jede Bewerbung ihre eigene Section-Reihenfolge
% (+ \clearpage-Positionen) bestimmen.
\IfFileExists{sections.tex}{%
  \input{sections.tex}%
}{%
  \markdownInput{projects.md}%
  \clearpage
  \IfFileExists{experience.md}{\markdownInput{experience.md}}{}%
  \clearpage
  \IfFileExists{languages.md}{\markdownInput{languages.md}}{}%
  \IfFileExists{skills.tex}{\subsection{Skills}\inputskills}{}%
}%

\end{document}
